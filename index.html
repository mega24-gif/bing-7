function doPost(e) { try { var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Hasil Belajar"); if (!sheet) { sheet = SpreadsheetApp.getActiveSpreadsheet().insertSheet("Hasil Belajar"); sheet.appendRow([ "Timestamp", "Nama", "Nickname", "Kelas", "No Absen", "Sekolah", "Nilai", "Benar", "Salah", "Persentase", "Waktu", "Achievement" ]); } var data = JSON.parse(e.postData.contents); sheet.appendRow([ new Date(), data.nama || "-", data.nickname || "-", data.kelas || "-", data.absen || "-", data.sekolah || "-", data.nilai || 0, data.benar || 0, data.salah || 0, data.persentase + "%", data.waktu || "-", data.achievement || "-" ]); return ContentService.createTextOutput(JSON.stringify({ success: true, message: "Result saved successfully" })).setMimeType(ContentService.MimeType.JSON); } catch (error) { return ContentService.createTextOutput(JSON.stringify({ success: false, message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } }
